async function findReportFolder(token){
 const site=await graphJson(
  'https://graph.microsoft.com/v1.0/sites/'+MS365.hostname+':'+MS365.sitePath+'?$select=id',
  token,
  'SharePoint site lookup'
 );

 const drives=await graphJson(
  'https://graph.microsoft.com/v1.0/sites/'+site.id+'/drives?$select=id,name',
  token,
  'Document library lookup'
 );

 const candidates=(drives.value||[]).filter(d=>!/teams wiki data/i.test(d.name));
 const drive=candidates.find(
  d=>/freigegebene dokumente|shared documents|documents|dokumente/i.test(d.name)
 );

 if(!drive){
  const names=(drives.value||[]).map(d=>d.name).join(', ');
  throw new Error(
   'Main SharePoint document library was not identified. Available libraries: '+names
  );
 }

 // Exact SharePoint destination for Pre-Trade Check reports.
 // This corresponds to the Windows-synced folder:
 // Research & Development - Monthly performance  Advisory 1\Pre_Trade_Check_Reports
 const reportPath=[
  'General',
  'Performance live',
  'Monthly performance  Advisory 1',
  'Pre_Trade_Check_Reports'
 ];

 let parentId='root';

 for(const part of reportPath){

  const url=parentId==='root'
   ? 'https://graph.microsoft.com/v1.0/drives/'+drive.id+
     '/root/children?$select=id,name,folder'
   : 'https://graph.microsoft.com/v1.0/drives/'+drive.id+
     '/items/'+parentId+'/children?$select=id,name,folder';

  const children=await graphJson(
   url,
   token,
   'Report folder lookup: '+part
  );

  const next=(children.value||[]).find(
   x=>x.folder &&
   norm(x.name).toLowerCase()===part.toLowerCase()
  );

  if(!next){
   const items=(children.value||[])
    .map(x=>(x.folder?'[folder] ':'[file] ')+x.name)
    .slice(0,50);

   throw new Error(
    'Report folder path failed at "'+part+
    '" in document library "'+drive.name+'". '+
    (items.length
     ? 'Items at this level: '+items.join(', ')
     : 'No child items were returned at this level.')
   );
  }

  parentId=next.id;
 }

 return{
  drive,
  folder:{
   id:parentId,
   name:'Pre_Trade_Check_Reports'
  }
 };
}

function complianceMonth(record){
 const value=String(record.check_date||record.allocation_date||'').trim();
 let year,month,day;
 let m=value.match(/^(20[0-9]{2})-([0-9]{2})-([0-9]{2})$/);
 if(m){year=Number(m[1]);month=Number(m[2]);day=Number(m[3]);}
 else{
  m=value.match(/^([0-9]{1,2})\/([0-9]{1,2})\/(20[0-9]{2})$/);
  if(m){year=Number(m[3]);month=Number(m[2]);day=Number(m[1]);}
 }
 if(!year||!month||!day)throw new Error('Cannot determine Compliance month from Combined Daily Allocation check date: '+value);
 const date=new Date(Date.UTC(year,month-1,day));
 if(date.getUTCFullYear()!==year||date.getUTCMonth()!==month-1||date.getUTCDate()!==day)throw new Error('Invalid Combined Daily Allocation check date: '+value);
 return year+'-'+String(month).padStart(2,'0')+'-'+['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'][month-1];
}
